# The main screen: the map with the panel under it, and beside both the
# sidebar (the minimap, the End turn and Undo buttons, the lists of cities,
# the buttons that go to the next stack and city, and the selected city).
#
# C++: assets/layouts/main.lua, which built the map, the minimap and a button
# from Lua. Session sizes the map and the minimap from the window
# (Session::apply_layout_sizes): the map is 80% of the width and all the
# height the panel under it leaves; the minimap is the rest of the width and
# 30% of the height. Session also makes the stack's slots, sizes the lines of
# text and writes them (src/info.rs has the words).
#
# Only the map takes the keyboard focus, so its keys work whatever was clicked.
TBLayout: id: main_layout, axis: x, distribution: gravity, distribution-position: left, position: top, spacing: 5
	TBLayout: id: map_column, axis: y, distribution-position: top, position: left, spacing: 5
		MapWidget: id: main_map, is-focusable: 1
		# Under the map: what is under the cursor; the selected hex; a slot
		# for each unit a hex can hold, in the stack's order (Session makes
		# them: `unit_0` ..., a button with the unit's icon and its strength
		# and moves, which selects the unit); and the selected unit in a line.
		TBLayout: id: hex_panel, axis: y, distribution-position: top, position: left, spacing: 3
			TBTextField: id: status, text-align: left
			TBTextField: id: hex_title, text-align: left
			TBLayout: id: stack, axis: x, distribution-position: left, position: top, spacing: 4
			TBTextField: id: unit_line, text-align: left
	TBLayout: id: main_sidebar, axis: y, distribution-position: top, spacing: 5
		MiniMapWidget: id: mini_map
		TBLayout: id: turn_row, axis: x, spacing: 5
			TBButton: id: end_turn, text: End turn, is-focusable: 0, squeezable: 1
			TBButton: id: undo, text: Undo, is-focusable: 0, squeezable: 1
		# The player's treasury: its gold, and its taxes and upkeep a turn.
		TBLayout: id: treasury, axis: y, distribution-position: top, position: left, spacing: 0
			TBTextField: id: gold_0, text-align: left
			TBTextField: id: gold_1, text-align: left
		# The lists of cities: every one, and the player's.
		TBLayout: id: city_lists, axis: x, spacing: 5
			TBButton: id: all_cities, text: All cities, is-focusable: 0, squeezable: 1
			TBButton: id: my_cities, text: My cities, is-focusable: 0, squeezable: 1
		# One after the other: the stacks that can still move, the cities.
		TBLayout: id: next_row, axis: x, spacing: 5
			TBButton: id: next_stack, text: Next stack, is-focusable: 0, squeezable: 1
			TBButton: id: next_city, text: Next city, is-focusable: 0, squeezable: 1
		# The selected units: send them somewhere over turns (pick the hex on
		# the map), or be done with them for this turn.
		TBLayout: id: order_row, axis: x, spacing: 5
			TBButton: id: go_to, text: Go to, is-focusable: 0, squeezable: 1
			TBButton: id: done, text: Done, is-focusable: 0, squeezable: 1
		# What the game's history says to the player: its battles, the cities
		# it took and lost; and what it has left to do this turn (Session
		# opens them).
		TBLayout: id: log_row, axis: x, spacing: 5
			TBButton: id: log, text: Log, is-focusable: 0, squeezable: 1
			TBButton: id: tasks, text: Tasks, is-focusable: 0, squeezable: 1
		# The city on the selected hex, a line at a time, and a button that
		# opens it in full (Session shows it when there is a city).
		TBLayout: id: city_panel, axis: y, distribution-position: top, position: left, spacing: 0
			TBTextField: id: city_0, text-align: left
			TBTextField: id: city_1, text-align: left
			TBTextField: id: city_2, text-align: left
			TBTextField: id: city_3, text-align: left
			TBTextField: id: city_4, text-align: left
			TBTextField: id: city_5, text-align: left
			TBTextField: id: city_6, text-align: left
			TBTextField: id: city_7, text-align: left
			TBTextField: id: city_8, text-align: left
		TBButton: id: city_details, text: City details, is-focusable: 0, visibility: gone
